多线程&多进程#

由于GIL（全局解释器锁），Python多线程无法真正并行（仅并发）。 多进程可绕过GIL，实现真正的并行（利用多核CPU）。 多进程：

import os
from multiprocessing import Process
def process_task(name):
print(f"子进程 {name} PID: {os.getpid()}，父进程 PID: {os.getppid()}")
if __name__ == '__main__':
p = Process(target=process_task, args=('child',))
p.start()
p.join()
print(f"主进程 PID: {os.getpid()}")

多线程

import threading
def thread_task(name):
print(f"线程 {name} 正在执行")
threads = []
for i in range(3):
t = threading.Thread(target=thread_task, args=(f'Thread-{i}',))
threads.append(t)
t.start()
for t in threads:
t.join()

线程池

from concurrent.futures import ThreadPoolExecutor
import requests
def fetch_url(url):
response = requests.get(url)
return f"{url}: {len(response.content)} bytes"
urls = ["https://www.python.org", "https://www.google.com", "https://www.github.com"]
# 使用线程池（适合I/O阻塞操作）
with ThreadPoolExecutor(max_workers=3) as executor:
results = list(executor.map(fetch_url, urls))
print(results)

锁仅针对于多线程

import threading
counter = 0
lock = threading.Lock()
def increment():
global counter
for _ in range(100000):
with lock:  # 自动获取和释放锁
counter += 1
threads = [threading.Thread(target=increment) for _ in range(5)]
for t in threads:
t.start()
for t in threads:
t.join()
print(f"最终计数器值: {counter}")  # 应该是500000

import pymysql
# 连接
conn = pymysql.connect(host='localhost', user='root', database='test_db')
try:
with conn.cursor() as cursor:
# 创建表
cursor.execute("""
CREATE TABLE IF NOT EXISTS users (
id INT AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(80),
age INT
)
""")
# 插入数据
cursor.execute("INSERT INTO users (name, age) VALUES (%s, %s)", ('Bob', 30))
conn.commit()
# 查询
cursor.execute("SELECT * FROM users")
print(cursor.fetchall())
finally:
conn.close()
